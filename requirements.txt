# Air Touch runtime dependencies (single source of truth).
# Used by bootstrap/install_runtime.py, `pip install -r requirements.txt`
# and (via pyproject.toml) `pip install -e .`
#
# MediaPipe MUST stay at 0.10.14 while the engine uses the legacy
# `mediapipe.solutions.hands` API (removed in newer MediaPipe releases).
#
# The installer installs this one line with `--no-deps`: MediaPipe 0.10.14 declares
# jax/jaxlib/scipy (~0.5 GB) that hand tracking never uses, and they make pip
# backtrack for minutes. Its REAL runtime needs are listed explicitly below, so a
# plain `pip install -r requirements.txt` works too (just heavier).
mediapipe==0.10.14

# --- what mediapipe.solutions.hands actually imports ---
absl-py>=2.0
attrs>=19.1.0
flatbuffers>=2.0
matplotlib>=3.7,<4          # imported at top level by mediapipe.solutions.drawing_utils
protobuf>=4.25.3,<5

# MediaPipe 0.10.14 depends on opencv-contrib-python; declaring it here (instead
# of opencv-python) avoids two packages fighting over the same `cv2` module.
# 0.10.14 is a NumPy 1.x build, so NumPy 2 is excluded.
opencv-contrib-python>=4.8,<4.12
numpy>=1.24,<2

# Desktop input adapter (Windows / Linux X11 / macOS)
PyAutoGUI>=0.9.54,<1

# Local developer console
